\item J. C. Baez, \emph{The Octonions}, Bull. Amer. Math. Soc. 39 (2002) 145–205, arXiv:math/0105155, section 2.2 (the page https://math.ucr.edu/home/baez/octonions/node5.html, which the task statement itself cites, was read on 2026-10-26): the Cayley–Dickson product $(a,b)(c,d)=(ac-db^{*},\ a^{*}d+cb)$, where Baez writes $x^{*}$ for the conjugate $\bar x$ and the conjugate of a pair is $(a,b)^{*}=(a^{*},-b)$. 7+1.nb proves that its octonions satisfy this product with $x=a+\ell b$, $\ell=e_4$.

\Needspace{8\baselineskip}\subsection*{2.1 Octonion matrices \texorpdfstring{$\tau^{O}$}{τO}}\addcontentsline{toc}{subsection}{2.1 Octonion matrices \texorpdfstring{$\tau^{O}$}{τO}}
Origin: the split matrices of 4+4.nb and the observation (checked) that $\sigma=\tau_1\tau_2\tau_3=\tau_4\tau_5\tau_6\tau_7$ commutes with $\tau_1,\tau_2,\tau_3$, anticommutes with $\tau_4,\dots,\tau_7$ and squares to $1_8$. Hence
\begin{align*}
\tau^{O}_0&=1_8,\quad \tau^{O}_A=\tau_A\ (A\le3),\quad \tau^{O}_k=\tau_k\sigma\ (k\ge4),\qquad (\tau_k\sigma)^2=\tau_k\sigma\tau_k\sigma=-\tau_k^2\sigma^2=-1_8, \\
\tau^{O}_h\tau^{O}_k&=\tau_h\tau_k\sigma=-\tau_k\tau_h\sigma=-\tau_k\sigma\tau_h=-\tau^{O}_k\tau^{O}_h,\qquad \tau^{O}_k\tau^{O}_l=\tau_k\sigma\tau_l\sigma=-\tau_k\tau_l=\tau_l\tau_k=-\tau^{O}_l\tau^{O}_k\quad(k\neq l\ge4).
\end{align*}
This is the real counterpart of the complex substitution $\tau_k\to i\tau_k$ that turns split octonions into octonions over $\mathbb C$.

\Needspace{8\baselineskip}\subsection*{2.2 Octonion vielbein and multiplication constants}\addcontentsline{toc}{subsection}{2.2 Octonion vielbein and multiplication constants}
Origin: In[294] and In[325] of 4+4.nb with the split spinor metric $\sigma$ replaced by the Euclidean $g^{O}=1_8$ and the same unit spinor $u$:
\begin{align*}
(E^{O})_{Ba}&=\big(u^{\mathsf T}g^{O}\bar\tau^{O}_B\big)_a,\qquad L^{O}_A=g^{O}_{AA}E^{O}\tau^{O}_A(E^{O})^{-1},\qquad e_Ae_K=\sum_L(L^{O}_A)_{LK}e_L .

Why the result is the octonion algebra: $(E^{O})^{-1}e_0=u$ makes $e_0$ a two-sided unit, and $\tau^{O}(x)\bar\tau^{O}(x)=N(x)1_8$ together with the orthogonality of $E^{O}$ gives $N(xy)=N(x)N(y)$. A unital composition algebra of dimension 8 with a positive definite norm is the octonions (Hurwitz). The notebook checks all the laws directly.

\Needspace{8\baselineskip}\subsection*{2.5 Real 7+1 generators \texorpdfstring{$\Gamma^{(1,7)}$}{Γ(1,7)}}\addcontentsline{toc}{subsection}{2.5 Real 7+1 generators \texorpdfstring{$\Gamma^{(1,7)}$}{Γ(1,7)}}
Origin: the standard construction of a Lorentzian Clifford algebra from a Euclidean one, applied to $T^{O}$:
\begin{align*}
\Gamma^{(1,7)}_0&=T^{O}_0,\qquad \Gamma^{(1,7)}_i=T^{O}_0T^{O}_i,\qquad (T^{O}_0T^{O}_i)^2=-(T^{O}_0)^2(T^{O}_i)^2=-1_{16}, \\
\lbrace T^{O}_0,T^{O}_0T^{O}_i\rbrace&=T^{O}_i+T^{O}_0T^{O}_iT^{O}_0=T^{O}_i-T^{O}_i=0,\qquad \lbrace T^{O}_0T^{O}_i,T^{O}_0T^{O}_j\rbrace=-\lbrace T^{O}_i,T^{O}_j\rbrace=-2\delta_{ij}1_{16}.
\end{align*}
Hence $\lbrace\Gamma^{(1,7)}_A,\Gamma^{(1,7)}_B\rbrace=2\eta^{(1,7)}_{AB}1_{16}$, and $\Gamma^{(1,7)}(x)^2=\mathrm{Re}(xx)1_{16}$.

\Needspace{8\baselineskip}\subsection*{2.9 Examples}\addcontentsline{toc}{subsection}{2.9 Examples}
so(1,7): $\Sigma_{AB}=\frac14[\Gamma^{(1,7)}_A,\Gamma^{(1,7)}_B]$; the commutation relations follow from the Clifford relation. Boost and rotation: $(\Gamma^{(1,7)}_0\Gamma^{(1,7)}_1)^2=1_{16}$, $(\Gamma^{(1,7)}_4\Gamma^{(1,7)}_5)^2=-1_{16}$, so $e^{\beta\Sigma_{01}}=\cosh\frac\beta2+\sinh\frac\beta2\Gamma^{(1,7)}_0\Gamma^{(1,7)}_1$ and $e^{\theta\Sigma_{45}}=\cos\frac\theta2+\sin\frac\theta2\Gamma^{(1,7)}_4\Gamma^{(1,7)}_5$. Dirac operator: $(\Gamma^A\partial_A)^2=\frac12\lbrace\Gamma^A,\Gamma^B\rbrace\partial_A\partial_B=\eta_{(1,7)}^{AB}\partial_A\partial_B$. Real amplitudes: $p=(3,2,2,0,0,0,0,0)$, $p^{\mathsf T}\eta^{(1,7)}p=1$, $\not p^{\,2}=1_{16}$; the real matrix $\not p$ has a real 8-dimensional eigenspace for the eigenvalue $+1$.

Real spinor fields. If $J$ anticommutes with every $\Gamma^{(1,7)}_A$, then $J\Gamma^{(1,7)}_8$ commutes with every $\Gamma^{(1,7)}_A$ (because $\Gamma^{(1,7)}_8$ anticommutes with each of them), so by irreducibility $J\Gamma^{(1,7)}_8=c\,1_{16}$ and $J=-c\,\Gamma^{(1,7)}_8$ (as $(\Gamma^{(1,7)}_8)^{-1}=-\Gamma^{(1,7)}_8$), whose square is $-c^2\,1_{16}$. Hence the real operator $\Gamma^A\partial_A-mJ$ has the square $\partial_0^2-\dots-\partial_7^2-c^2m^2$ (a tachyon), and a real equation $(G^A\partial_A-m)\Psi=0$ implies the Klein–Gordon equation only if the real $G_A$ obey the $\eta^{(7,1)}$ relation, impossible for 16×16 matrices (2.7). Massless: for $k=(1,1,0,\dots,0)$, $\not k^{\,2}=k^{\mathsf T}\eta^{(1,7)}k\,1_{16}=0$ and $\Gamma^A\partial_A[w\cos(k\cdot x)]=-\sin(k\cdot x)\,\not k\,w$, which vanishes for the 8 real solutions of $\not k\,w=0$. Massive, with the real 32×32 matrices: $P=\sum_Ap^A\widehat\Gamma^{(7,1)}_A$, $P^2=-m^2\,1_{32}$, and with $\phi=p\cdot x$, $\widehat\Gamma^A=\eta_{(7,1)}^{AA}\widehat\Gamma^{(7,1)}_A$:
\begin{align*}
\widehat\Gamma^{A}\partial_A\Big(w\cos\phi+\tfrac1mPw\sin\phi\Big)&=-P\Big(-w\sin\phi+\tfrac1mPw\cos\phi\Big) \\
&=Pw\sin\phi+m\,w\cos\phi=m\Big(w\cos\phi+\tfrac1mPw\sin\phi\Big),
\end{align*}
using $\partial_A\phi=(\eta^{(1,7)}p)_A$ and $\eta_{(7,1)}^{AA}\eta^{(1,7)}_{AA}=-1$.

\textbf{Dirac matrices.} The real generators $\Gamma^{(1,7)}_A$ of section 2.5 are used unchanged: $\Gamma_0=\left(\begin{smallmatrix}0&1_8\\1_8&0\end{smallmatrix}\right)$, $\Gamma_i=\left(\begin{smallmatrix}\tau^{O}_i&0\\0&-\tau^{O}_i\end{smallmatrix}\right)$, $\Gamma_A\Gamma_B+\Gamma_B\Gamma_A=2\eta^{(1,7)}_{AB}1_{16}$. \end{align*}
\textbf{Real amplitudes.} For $e^{-i(Ex_0-\vec p\cdot\vec x)}$, with $P_\tau=p^i\tau^{O}_i$ ($P_\tau^{\mathsf T}=-P_\tau$, $P_\tau^2=-\vec p^{\,2}1_8$):
\begin{align*}
H_{\rm D}(\vec p)&=\begin{pmatrix}0&m1_8+P_\tau\\m1_8-P_\tau&0\end{pmatrix},\qquad (m1_8+P_\tau)(m1_8-P_\tau)=(m^2+\vec p^{\,2})1_8=E^21_8, \\

\textbf{Chirality.} The real matrices anticommuting with every $\Gamma_A$ are $c\,\Gamma^{(1,7)}_8$ (section 2.9), with square $-c^21_{16}$; the chirality is therefore the complex $\chi=i\Gamma^{(1,7)}_8=\left(\begin{smallmatrix}0&-i1_8\\i1_8&0\end{smallmatrix}\right)$. With $E_\pm$ the 16×8 matrices $\frac1{\sqrt2}\left(\begin{smallmatrix}1_8\\-i1_8\end{smallmatrix}\right)$ and $\frac1{\sqrt2}\left(\begin{smallmatrix}-i1_8\\1_8\end{smallmatrix}\right)$ ($\chi E_-=-E_-$, $\chi E_+=E_+$, $E_-E_-^\dagger=P^{(1,7)}_L$, $E_+E_+^\dagger=P^{(1,7)}_R$), $\Psi_L=E_-^\dagger\Psi$, $\Psi_R=E_+^\dagger\Psi$ and

\textbf{Self-coupling.} The four Lorentz-invariant contact interactions without derivatives used in relativistic point-coupling models (scalar, pseudoscalar, vector, axial vector) with real couplings, and real sources that define the densities:
\begin{align*}

&\quad-\frac{g_S}2(\bar\Psi\Psi)^2-\frac{g_P}2(\bar\Psi i\chi\Psi)^2-\frac{g_V}2(\bar\Psi\Gamma^{A}\Psi)(\bar\Psi\Gamma_A\Psi)-\frac{g_A}2(\bar\Psi\Gamma^{A}\chi\Psi)(\bar\Psi\Gamma_A\chi\Psi).

The tensor and higher-rank channels are not included. With $\hbar=c=1$ every $g$ has mass dimension $-6$ (an effective theory: no-sea approximation, or a momentum cutoff for the sea). \textbf{Hohenberg–Kohn.} For $H[w]=H_0+H_{\rm int}+\int w\cdot n$ with $w\cdot n=\mathsf s\rho_S+\mathsf p\rho_P+\mathsf v_Aj^A+\mathsf a_Aj_5^A$, two sources $w\neq w'$ with different ground states and the same densities would give $E<E'+\int(w-w')\cdot n$ and $E'<E+\int(w'-w)\cdot n$, so $E+E'<E+E'$; hence $E[w]=\min_n(F[n]+\int w\cdot n)$. \textbf{Kohn–Sham.} Varying $T_s+\int w\cdot n+E_H+E_{xc}$ with respect to $\Phi_k^\dagger$ under $\int\Phi_k^\dagger\Phi_l=\delta_{kl}$ (multipliers $\lambda_k$), with $\delta\rho_S/\delta\Phi^\dagger=\Gamma^0\Phi$, $\delta\rho_P/\delta\Phi^\dagger=\Gamma^0i\chi\Phi$, $\delta j^A/\delta\Phi^\dagger=\Gamma^0\Gamma^A\Phi$, $\delta j_5^A/\delta\Phi^\dagger=\Gamma^0\Gamma^A\chi\Phi$, gives $H_{\rm KS}\Phi_k=\lambda_k\Phi_k$ with
\begin{align*}
H_{\rm KS}&=\Gamma^0\Gamma^i(-i\partial_i+\mathcal V_i)+\Gamma^0(m+\mathcal S)+\mathcal V_0+\Gamma^0i\chi\mathcal P+\chi\mathcal A_0+\Gamma^0\Gamma^i\chi\mathcal A_i, \\
\mathcal S&=\mathsf s+g_S\rho_S+\frac{\delta E_{xc}}{\delta\rho_S},\qquad \mathcal P=\mathsf p+g_P\rho_P+\frac{\delta E_{xc}}{\delta\rho_P}, \\
\mathcal V_A&=\mathsf v_A+g_Vj_A+\frac{\delta E_{xc}}{\delta j^A},\qquad \mathcal A_A=\mathsf a_A+g_Aj_{5A}+\frac{\delta E_{xc}}{\delta j_5^A},
\end{align*}
where $E_H=\int[\frac{g_S}2\rho_S^2+\frac{g_P}2\rho_P^2+\frac{g_V}2j^Aj_A+\frac{g_A}2j_5^Aj_{5A}]$ and $T_s=\sum_k\int\Phi_k^\dagger H_{\rm D}\Phi_k$. \textbf{Decision (no sea):} the orbitals are the occupied positive-energy states. \textbf{Exchange.} Wick’s theorem for a Slater determinant, $\langle c_a^\dagger c_c^\dagger c_dc_b\rangle=\varrho_{ba}\varrho_{dc}-\varrho_{da}\varrho_{bc}$ with $\varrho_{xy}=\sum_k\Phi_{k,x}\Phi_{k,y}^*$ and $\mathcal R=\varrho\,\Gamma^0$, gives $\langle:(\bar\Psi O_1\Psi)(\bar\Psi O_2\Psi):\rangle=\mathrm{tr}(O_1\mathcal R)\mathrm{tr}(O_2\mathcal R)-\mathrm{tr}(O_1\mathcal RO_2\mathcal R)$, hence the exact exchange energy

E_x=-\frac{g_S}2\mathrm{tr}(\mathcal R\mathcal R)-\frac{g_P}2\mathrm{tr}(i\chi\mathcal R\,i\chi\mathcal R)-\frac{g_V}2\mathrm{tr}(\Gamma^{A}\mathcal R\Gamma_A\mathcal R)-\frac{g_A}2\mathrm{tr}(\Gamma^{A}\chi\mathcal R\Gamma_A\chi\mathcal R).
\end{equation*}

e^{i\vartheta\chi}&=\cos\vartheta\,1_{16}-\sin\vartheta\,\Gamma^{(1,7)}_8:\qquad \kappa\to e^{2i\vartheta}\kappa, \\
\Delta E_H&=\frac{g_S-g_P}2\big[(e^{4i\vartheta}-1)\kappa^2+(e^{-4i\vartheta}-1)\kappa^{*2}\big].
\end{align*}
\textbf{Uniform matter.} Units $m=1$; all 8 spin states filled to $k_F$; $|S^6|=16\pi^3/15$ is the area of the unit sphere in $\mathbb R^7$ and $c_7=8|S^6|/(2\pi)^7=1/(15\pi^4)$. With the energy density $\mathcal E$, its kinetic part $\mathcal E_{\rm kin}$, $\rho_S$ and the density $\rho$:
\begin{align*}
\rho&=\frac{k_F^7}{105\pi^4},\qquad \rho_S=c_7\int_0^{k_F}\frac{k^6m^*\,dk}{\sqrt{k^2+m^{*2}}},\qquad \mathcal E_{\rm kin}=c_7\int_0^{k_F}k^6\sqrt{k^2+m^{*2}}\,dk, \\
m^*&=m+g_S\rho_S(m^*),\qquad \mathcal E=\mathcal E_{\rm kin}-\frac{(m^*-m)^2}{2g_S}+\frac{g_V}2\rho^2, \\
E/A&=\frac{\mathcal E}{\rho},\qquad E_F=g_V\rho+\sqrt{k_F^2+m^{*2}} .
\end{align*}
Because $\partial\mathcal E/\partial m^*=0$ is the gap equation, $d\mathcal E/d\rho=E_F$ at every density, and $\frac{d}{dk_F}\frac{\mathcal E}\rho=\frac{c_7k_F^6}{\rho}(E_F-\frac{\mathcal E}{\rho})$ vanishes at saturation: $E/A=E_F$ (Hugenholtz–Van Hove). \textbf{Decisions:} $g_S=-10^5$, $g_V=10^4$, $g_P=g_A=0$ (binding and saturation); root finding by Brent’s method with 50-digit (gap equation) and 40-digit (saturation) working precision. \textbf{Dirac sea.} With a cutoff $\Lambda_{\rm c}$, $g_S=g_P=g$ and a uniform $\mathcal M$:
\begin{align*}
J(M)&=\frac1{120\pi^4}\int_0^{\Lambda_{\rm c}}\frac{q^6dq}{\sqrt{q^2+M^2}},\qquad K(M)=\frac1{120\pi^4}\int_0^{\Lambda_{\rm c}}q^6\sqrt{q^2+M^2}\,dq, \\
\kappa&=-4\mathcal M^*J(|\mathcal M|),\qquad \mathcal M=m-8gJ(|\mathcal M|)\mathcal M,\qquad g_c=-\frac1{8J(0)}=-\frac{90\pi^4}{\Lambda_{\rm c}^6}, \\
\mathcal E_{\rm vac}&=-8K+8J\,\mathrm{Re}[(\mathcal M-m)\mathcal M^*]+32g|\mathcal M|^2J^2 .
\end{align*}
\textbf{Decisions:} $\Lambda_{\rm c}=1$, $g=\frac65g_c$, and $m=\frac1{100}$ for the explicit-breaking example.
